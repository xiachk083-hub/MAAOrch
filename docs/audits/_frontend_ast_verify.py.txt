def ast_routes(text, label=""):
    tree = ast.parse(text)
    out, skipped = [], []
    for node in ast.walk(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        for dec in node.decorator_list:
            if not isinstance(dec, ast.Call) or not isinstance(dec.func, ast.Attribute):
                continue
            attr = dec.func.attr.lower()
            if attr not in HTTP:
                if attr == "middleware":
                    skipped.append(("middleware", node.name))
                continue
            if not (isinstance(dec.func.value, ast.Name) and dec.func.value.id == "app"):
                skipped.append(("non-app." + attr, node.name)); continue
            if dec.args and isinstance(dec.args[0], ast.Constant) and isinstance(dec.args[0].value, str):
                out.append((attr.upper(), dec.args[0].value))
            else:
                skipped.append(("non-literal", node.name))
    return out, skipped

def const_str(node):
    if isinstance(node, ast.Constant) and isinstance(node.value, str):
        return node.value
    if isinstance(node, ast.JoinedStr):
        parts = []
        for v in node.values:
            if isinstance(v, ast.Constant) and isinstance(v.value, str):
                parts.append(v.value)
            elif isinstance(v, ast.FormattedValue):
                parts.append("{" + ast.unparse(v.value) + "}")
        return "".join(parts)
    if isinstance(node, ast.BinOp) and isinstance(node.op, ast.Add):
        l = const_str(node.left)
        if l is not None:
            r = const_str(node.right)
            return l + (r if r is not None else "{}")
    return None
