# -*- coding: utf-8 -*-
"""task-15：500 -> 404 的独立复现（农场法：同一份现网文件布局，只换 api_fastapi.py 版本）。
before = 退役前 9f9fab9^ 的 api_fastapi.py（catch-all 指向已删除的 ui/web/index.html）
after  = HEAD 的 api_fastapi.py
跑法：python docs/audits/_frontend_500_probe.py.txt
"""
import json, os, shutil, subprocess, sys, tempfile
from pathlib import Path

SRC = Path(__file__).resolve().parents[2]
PROBE = SRC / "docs" / "audits" / "_farm_probe.py.txt"

def build_farm(root):
    root.mkdir(parents=True, exist_ok=True)
    for e in SRC.iterdir():
        if e.name in {".git"}:
            continue
        dst = root / e.name
        if e.is_dir():
            if e.name in {"network", "tests"}:
                continue
            subprocess.run(["cmd", "/c", "mklink", "/J", str(dst), str(e)], capture_output=True, text=True)
        else:
            try: os.link(e, dst)
            except OSError: pass
    for pkg in ("network", "tests"):
        (root / pkg).mkdir(exist_ok=True)
        for e in (SRC / pkg).iterdir():
            if e.is_file():
                try: os.link(e, root / pkg / e.name)
                except OSError: pass
    shutil.copy2(PROBE, root / "_farm_probe.py")

old_src = subprocess.run(["git", "--no-optional-locks", "show", "9f9fab9^:network/api_fastapi.py"],
                         capture_output=True).stdout.decode("utf-8")
new_src = (SRC / "network" / "api_fastapi.py").read_text(encoding="utf-8")
print("ui/web 是否存在（现网布局）: %s" % (SRC / "ui" / "web").exists())

work = Path(tempfile.mkdtemp(prefix="task15_500_", dir=str(SRC.parent)))
try:
    build_farm(work)
    for tag, src in (("before(9f9fab9^)", old_src), ("after(HEAD)", new_src)):
        (work / "network" / "api_fastapi.py").write_text(src, encoding="utf-8", newline="")
        r = subprocess.run([sys.executable, "_farm_probe.py"], cwd=str(work),
                           capture_output=True, text=True, timeout=300)
        out = (r.stdout or "").strip().splitlines()
        print("  %-16s rc=%d  %s" % (tag, r.returncode, out[-1] if out else (r.stderr or "")[-300:]))
        if r.returncode != 0 and r.stderr:
            print("      stderr: %s" % r.stderr.strip().splitlines()[-1][:200])
finally:
    shutil.rmtree(work, ignore_errors=True)
    print("农场已删除")
