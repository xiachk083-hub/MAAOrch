# 2) G5 守卫短路推演
print("\n===== G5 release 顺序 vs on_account_finished 守卫 =====")
def mimic(order):
    lq = {"active_emus": {"emu0": "A1"}, "ts": {"emu0": 1.0}}
    def release():
        lq["active_emus"].pop("emu0", None); lq["ts"].pop("emu0", None)
    def hook(aid):
        old = lq["active_emus"].pop("emu0", None)
        lq["ts"].pop("emu0", None)
        return "early-return(重试/退避不可达)" if old != aid else "proceed(重试/退避可达)"
    if order == "release-first":
        release(); return hook("A1")
    r = hook("A1"); release(); return r
print("  release-first ->", mimic("release-first"))
print("  release-after ->", mimic("release-after"))
ok(mimic("release-first").startswith("early-return"), "改前顺序确实短路 on_account_finished（守卫 old != aid）")
ok(mimic("release-after").startswith("proceed"), "改后顺序守卫不再短路")

# 3) G5 双入队：enqueue 去重键(account_id, slot)
print("\n===== G5 双入队条件 =====")
def enq(pending, aid, slot):
    pending = [e for e in pending if not (e[0] == aid and e[1] == slot)]
    pending.append((aid, slot)); return pending
for tag, s in (("定时/手动/理智 slot=''", ""), ("force slot='fight'", "fight"), ("dispatch slot='roguelike'", "roguelike")):
    p = enq(enq([], "A1", ""), "A1", s)
    ok(len(p) == (1 if s == "" else 2), "%s → pending=%d %s" % (tag, len(p), p))
