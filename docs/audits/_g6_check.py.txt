# -*- coding: utf-8 -*-
"""G6 端到端：mcp_mount_enabled=False 时 create_app 是否跳过 fastmcp 导入。"""
import sys, time, os
sys.path.insert(0, os.getcwd())

class _Any:
    def __call__(self, *a, **k): return None
    def __getattr__(self, n): return _Any()
    def __iter__(self): return iter(())
    def __bool__(self): return False

class M:
    def __init__(self, on):
        self.config = {"mcp_mount_enabled": on}
        self.accounts = []
        self._log = lambda *a, **k: None
    def __getattr__(self, n): return _Any()

flag = sys.argv[1] == "on"
t0 = time.perf_counter()
from network import api_fastapi
t_imp = time.perf_counter() - t0
app = None
err = ""
try:
    t1 = time.perf_counter()
    app = api_fastapi.create_app(M(flag))
    t_create = time.perf_counter() - t1
except Exception as e:
    t_create = -1.0
    err = "%s: %s" % (type(e).__name__, str(e)[:200])
mounts = []
if app is not None:
    try:
        mounts = [getattr(r, "path", "?") for r in app.routes if getattr(r, "path", "").startswith("/mcp")]
    except Exception:
        pass
print("flag=%s  api_fastapi_import=%.2fs  create_app=%.2fs  fastmcp_in_sys.modules=%s  mcp_routes=%s  err=%s" % (
    "on" if flag else "off", t_imp, t_create, "fastmcp" in sys.modules, mounts, err))
