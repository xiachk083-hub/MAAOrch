print("\n===== T8-1 双入队（真实 LaunchQueue.enqueue + on_account_finished）=====")
def build_lq(mod, launch_slot, emu_idx="0"):
    ac = {"id": "A1", "name": "n1", "emu_instance_index": emu_idx,
          "consecutive_failures": 1, "_slot": launch_slot}
    ctx = types.SimpleNamespace(accounts=[ac], config={}, log=lambda m: None, _mw=None)
    lq = mod.LaunchQueue(ctx)
    lq._save_queue = lambda: None
    lq._has_running_process = lambda aid: False
    lq._tick = lambda: None
    lq._active_emus = {"0": "A1"} if emu_idx else {}
    lq._active_emus_ts = {"0": 1.0} if emu_idx else {}
    return lq, ac
def two_producers(mod, launch_slot, exit_code):
    lq, ac = build_lq(mod, launch_slot)
    lq.on_account_finished(("A1", exit_code, []))                   # 生产者① 队列重试（L350/373/398）
    lq.enqueue("A1", "auto", priority=0, slot=ac.get("_slot", ""))  # 生产者② runner 自动重启（L3248）
    return [(e.account_id, e.slot, e.source) for e in lq._pending]

        ok(len(new) == 1, "改造后 slot=%-6r %s → pending %d" % (slot, tag, len(new)))

# ═══ 2) T8：emit_finished 逐订阅者隔离 ═══
print("\n===== T8-2 emit_finished 订阅者异常隔离（爆炸订阅者桩）=====")
def emit_case(mod, tag):
    seen, warned = [], []
    def boom(data): raise RuntimeError("订阅者爆炸")
    def good(data): seen.append(data)
    stub = types.SimpleNamespace(_finished_callbacks=[boom, good],
                                 _log=types.SimpleNamespace(warn=lambda m: warned.append(m)))
    propagated = None
    try: mod.AccountRunner.emit_finished(stub, ("A1", 1, []))
    except Exception as e: propagated = e
    released = ["A1"] if propagated is None else []
    print("  %s propagated=%r good_ran=%s released_after=%s warn=%s" % (
        tag, propagated, seen == [("A1", 1, [])], released, bool(warned)))
    return propagated, seen, released, warned
r_old = load(base["services/runner.py"], "runner_old")
r_new = load(cur["services/runner.py"], "runner_new")

ok(s_old == [], "改造前：异常中断循环 → 后续正常订阅者收不到通知")
p_new, s_new, rel_new, w_new = emit_case(r_new, "改造后")

ok(s_new == [("A1", 1, [])], "改造后：其余订阅者照常收到通知")
ok(bool(w_new), "改造后：异常有 warn 留痕: %s" % (w_new[:1],))

print("\n===== T9 六处杀进程闸门（旧 vs 新差分）=====")
class R:
    def __init__(self, out=""): self.stdout = out; self.stderr = ""; self.returncode = 0
def harness(mod, gate, mw_config=None, lq_cfg=None, lq_missing=False, lq_cfg_none=False):
    tmp = Path(tempfile.mkdtemp(prefix="task11_inst_"))
    (tmp / "1").mkdir(); (tmp / "1" / ".pid").write_text("555"); (tmp / "1" / ".meta").write_text("A2|n2")
    (tmp / "2").mkdir(); (tmp / "2" / ".pid").write_text("666")
    calls = []
    def fake_run(cmd, *a, **k):
        c = list(cmd) if not isinstance(cmd, str) else [cmd]; calls.append(c)
        j = " ".join(str(x) for x in c)
        if "Measure-Object).Count" in j: return R("7")
        if "MAA.exe" in j and "CSV" in j: return R(chr(34) + "MAA.exe" + chr(34) + "," + chr(34) + "777" + chr(34))
        if "MuMuNxCrashReporter.exe" in j and "tasklist" in j: return R("MuMuNxCrashReporter.exe")
        return R("")
    class P:
        def __init__(self, pid, idx):
            import time
            self.info = {"name": "MuMuVMMHeadless.exe", "cmdline": ["x", "MuMuPlayer-12.0-" + idx],
                         "pid": pid, "create_time": time.time() - 1000}
    import psutil
    mod.INSTANCES = tmp; mod._pid_alive = lambda pid: True; mod._running_emulators = lambda: {}
    psutil.process_iter = lambda attrs=None: iter([P(111, "0"), P(222, "9")])
    class MW:
        def __init__(self):
            self.accounts = [{"id": "A1", "name": "n1", "emu_instance_index": "0"}]
            self.runner = None; self.config = mw_config if mw_config is not None else {"emu_reclaim_enabled": gate}
            self._msgs = []
        def _log(self, m): self._msgs.append(m)
    mw = MW()
    if not lq_missing:
        cfg = None if lq_cfg_none else (lq_cfg if lq_cfg is not None else {"emu_reclaim_enabled": gate})
        mw.launch_queue = types.SimpleNamespace(ctx=types.SimpleNamespace(config=cfg),
                                                _active_emus={"0": "A1"}, _pending=[])
    real = subprocess.run; subprocess.run = fake_run
    try: out = mod.check_health(mw)
    finally: subprocess.run = real
    shutil.rmtree(tmp, ignore_errors=True)
    kills = [c for c in calls if (c and "taskkill" in str(c[0])) or "Stop-Process" in " ".join(map(str, c))]
    return kills, out, mw._msgs
