# -*- coding: utf-8 -*-
"""task-15 变异测试：把白名单路由注释掉，验证 tests/test_routes_whitelist.py 真的会失败。
不碰原仓库文件：在 %TEMP% 建 hardlink/junction 农场，只在农场里做变异。
跑法：python docs/audits/_frontend_mutant_test.py.txt
"""
import os, re, shutil, subprocess, sys, tempfile
from pathlib import Path

SRC = Path(__file__).resolve().parents[2]
KEEP_REAL = {"network", "tests"}
SKIP = {".git"}

def build_farm(root):
    root.mkdir(parents=True, exist_ok=True)
    for e in SRC.iterdir():
        if e.name in SKIP:
            continue
        dst = root / e.name
        if e.is_dir():
            if e.name in KEEP_REAL:
                continue
            r = subprocess.run(["cmd", "/c", "mklink", "/J", str(dst), str(e)],
                               capture_output=True, text=True)
            if r.returncode != 0:
                raise RuntimeError("junction failed %s: %s" % (e.name, r.stdout + r.stderr))
        else:
            try:
                os.link(e, dst)
            except OSError as ex:
                print("  (跳过链接 %s: %s)" % (e.name, ex))
    for pkg in KEEP_REAL:
        (root / pkg).mkdir(exist_ok=True)
        for e in (SRC / pkg).iterdir():
            if e.is_dir():
                continue
            os.link(e, root / pkg / e.name)
    return root / "network" / "api_fastapi.py"

def run_pytest(root, what):
    r = subprocess.run([sys.executable, "-m", "pytest", "tests/test_routes_whitelist.py", "-q"],
                       cwd=str(root), capture_output=True, text=True, timeout=600)
    out = (r.stdout or "") + (r.stderr or "")
    tail = [l for l in out.splitlines() if l.strip()][-3:]
    failed = sorted({l.split(" ")[1] for l in out.splitlines() if l.startswith("FAILED ")})
    print("  [%s] rc=%d | %s" % (what, r.returncode, " / ".join(tail)))
    if failed:
        print("      FAILED: %s" % failed)
    return r.returncode, failed, out

base_src = (SRC / "network" / "api_fastapi.py").read_text(encoding="utf-8")
# 农场必须与仓库同盘（hardlink 不能跨盘）：建在仓库父目录，跑完删除
work = Path(tempfile.mkdtemp(prefix="task15_farm_", dir=str(SRC.parent)))
try:
    mutant_path = build_farm(work)
    print("农场: %s" % work)
    mutant_path.write_text(base_src, encoding="utf-8", newline="")
    rc0, f0, _ = run_pytest(work, "baseline 未变异")
    print("== baseline 期望 71 passed / rc=0: %s" % ("OK" if rc0 == 0 else "异常"))

    MUTATIONS = [
        ("M1 注释掉 POST /api/queue/pause（白名单+MCP queue_control）",
         lambda s: s.replace('    @app.post("/api/queue/pause")',
                             '    # MUTANT REMOVED @app.post("/api/queue/pause")', 1)),
        ("M2 注释掉 GET /api/health（白名单+MCP health）",
         lambda s: s.replace('    @app.get("/api/health")',
                             '    # MUTANT REMOVED @app.get("/api/health")', 1)),
        ("M3 新增白名单外路由 GET /api/zzz_extra",
         lambda s: s.replace('    @app.get("/api/status")',
                             '    @app.get("/api/zzz_extra")\n    def zzz_extra():\n        return {}\n\n'
                             '    @app.get("/api/status")', 1)),
        ("M4 把 GET /api/status 改成 POST /api/status（方法灵敏度）",
         lambda s: s.replace('    @app.get("/api/status")', '    @app.post("/api/status")', 1)),
        ("M5 把 /api/inspect/account/{name} 参数改名 {account_name}（字面量脆性）",
         lambda s: s.replace('@app.get("/api/inspect/account/{name}")',
                             '@app.get("/api/inspect/account/{account_name}")', 1)),
    ]
    results = {}
    for name, mut in MUTATIONS:
        s = mut(base_src)
        changed = s != base_src
        mutant_path.write_text(s, encoding="utf-8", newline="")
        rc, failed, out = run_pytest(work, name)
        results[name] = (changed, rc, failed)
        if not changed:
            print("      ⚠ 变异未能应用（锚点未命中）")
    print("\n=== 变异结果汇总 ===")
    for name, (changed, rc, failed) in results.items():
        print("  %-58s 变异生效=%-5s rc=%d 失败断言=%d %s" % (
            name[:56], changed, rc, len(failed), failed[:3]))
finally:
    shutil.rmtree(work, ignore_errors=True)
    print("农场已删除")
