# -*- coding: utf-8 -*-
import json
from fastapi.testclient import TestClient
from network.api_fastapi import create_app
class MW:
    config = {"api_token": "", "mcp_mount_enabled": False}
    def _log(self, *a, **k): pass
app = create_app(MW())
out = {}
with TestClient(app, raise_server_exceptions=False) as c:
    for p in ["/", "/index.html", "/ui/web/index.html", "/unknown/xyz", "/api/nope"]:
        try:
            out[p] = c.get(p).status_code
        except Exception as e:
            out[p] = "EXC:" + type(e).__name__
print(json.dumps(out, ensure_ascii=False))
